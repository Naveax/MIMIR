        return Err("payload-start-mismatch");
    }
    let tag_name = format!("{tag:?}");
    if row.attribute_tag.as_deref() != Some(tag_name.as_str()) {
        return Err("payload-tag-mismatch");
    }
    if (version_major, version_minor, net_version, is_rl_223)
        != (row.version_major, row.version_minor, row.net_version, row.is_rl_223)
    {
        return Err("payload-context-mismatch");
    }
    Ok(())
}

fn flip_bit(bytes: &mut [u8], bit: u64) {
    let index = usize::try_from(bit).expect("bit fits usize");
    if index / 8 < bytes.len() {
        bytes[index / 8] ^= 1u8 << (index % 8);
    }
}
